#### 0.4.3 复核中新发现（可复算，CACHE-MODEL 未量化）

1. **facts 快照只追加、永不替换 → 长会话线性累积**：after 的 7 条快照全部 `surfaceOp:append`
   （事件 seq 10/21/35/49/63/77/91），12 步共累积 **3213 字符**；prompt tokens 因此从 16278 涨到 19232
   （+2954 / 6 轮 ≈ +590 tok/轮，含正常对话）。每步只要事实变一次就 +458 字符（≈130–150 tok），
   工具循环多的会话（MC）会按步数累积。**建议**：CACHE-MODEL §5.4 把它从"略增"改成"线性累积 + 上界估算"，
   并考虑让 DSH 侧 replace 旧快照（语义上本来就是 supersedes），或 ICL 把快照节流到"每轮一次"。
2. **"方案 A 收益=0" 是引擎相关的结论，不能外推**：证据（stable/tailclock/headclock 三组）说明
   ninfer 的会话复用粒度是**整段 system**——system 任何差异都让 session digest 变、`hit=0`。
   按 token 前缀复用的引擎（如 llama.cpp 常见 `--cache-reuse`）通常能保住差异点之前的 tok，
   所以"把动态段挪到 system 尾部"在别的引擎上可能仍有一半左右收益。**建议** CACHE-MODEL §4.2 加一句引擎限定。

### 0.5 「facts 快照每轮只刷一次」节流的测量口径（设计给 cache；本轮未实现、未起 lab）

#### 0.5.1 先明确问题域（避免白做）

用 `cache-af2` 的落盘数据反查：**当前 12 步里只有 7 条快照 ≈ 1 条/轮**——因为 DSH 自己会去重
（`RuntimeContextProjection.project()` 在快照文本相同时返回 `undefined`，不 append）。也就是说
**"每步都 append" 不是现状**，只有"一个 turn 内事实每步都变"时才会发生；典型就是 MC 会话：
`mc:status`（order 900，位置/血量/背包）在 facts 里，每步都可能变。

先用 af2 实测换算收益上界（复算命令见 0.5.6）：

| 单次快照 append 的代价 | 实测（af2，`cache-facts-6`，2 步/轮） |
|---|---|
| 快照文本 | **458 字符 / 510 字节**（首条 465） |
| 该请求 prefill | 有快照的请求 191–194 tok，无快照的 49–50 tok → **净增 ≈142 tok** |
| 该轮 prompt 增量 | 有快照的步 +244、无快照的步 +51 → **净增 ≈190 tok 上下文** |

#### 0.5.4 实现选项（按推荐度）

- **Option 0（推荐先做，收益/风险最优）**：把高频环境事实移出 facts —— `mc:status`（900）不进 `icl:facts` 快照，
  她要位置/血量就调 `mcb_status`。MC 的每步 churn 直接归零，代价是每轮多一次工具调用（本来就在调 `mcb_do`）。
- **Option 1（若一定要放快照）**：turn 级快照 + 事件驱动 dirty。bridge 缓存 `factsText`，`sameTurn && !dirty` 时返回缓存；
  `dirty` 由 focus/time 的 `onChange`、`schedule-start`/`sleep-start` 事件置位。→ 39 步/turn 降到 1+事件数，
  且"她刚做过的事"（建焦点/排日程/睡醒）下一步可见。

  新安排要等下一轮才进事实，语义回归风险最大。

```js
// 用法：把本文件存成 %TEMP%\icl-audit-measure.mjs，然后
//   node %TEMP%\icl-audit-measure.mjs C:\DSH\icl-wt\<你的 worktree>
// 输出：一份 JSON（字段见 docs/AUDIT-2026-11-05.md §9.3 的"期望输出"）。
import fs from 'node:fs'
import os from 'node:os'
import path from 'node:path'
import { pathToFileURL } from 'node:url'

const root = process.argv[2] || process.cwd()
const { createHost } = await import(pathToFileURL(path.join(root, 'packages/core/src/index.mjs')).href)
const { normalizeConfig, DEFAULT_PLUGINS } = await import(pathToFileURL(path.join(root, 'packages/dsh-icl/src/config.mjs')).href)
const { loadPlugin } = await import(pathToFileURL(path.join(root, 'packages/dsh-icl/src/bootstrap.mjs')).href)

const bytes = (s) => Buffer.byteLength(String(s ?? ''), 'utf8')
const chars = (s) => String(s ?? '').length
const out = {}
const stateDir = fs.mkdtempSync(path.join(os.tmpdir(), 'icl-audit-'))
const T0 = Date.parse('2026-11-05T20:07:00+08:00')

const cfg = normalizeConfig({
  stateDir,
  personaFile: path.join(root, 'prompts/persona.md'),
  simulationFile: path.join(root, 'prompts/sim.md'),
  dreamPersonaFile: path.join(root, 'prompts/dream.md'),
  providers: [],
})
const host = createHost({ log: () => {}, now: () => T0 })
host.services.clock.set(() => T0)
const configFor = (id) => {
  switch (id) {
    case 'memory': return { root: stateDir, path: path.join(stateDir, 'state/memory.db') }
    case 'recall': return { sessionId: 'live' }
    case 'persona': return { card: cfg.persona }
    case 'time': return { stateFile: path.join(stateDir, 'state/time.json'), tickMs: 15_000 }
    case 'dream': return { sessionId: 'live', persona: cfg.dreamPersona, dreamAt: '23:00', timezone: 'Asia/Hong_Kong' }
    case 'mc-link': return { baseUrl: 'http://127.0.0.1:54836', logFile: path.join(stateDir, 'mc-link.jsonl'), pollMs: 120_000 }
    case 'model-manager': return { providers: [] }
    case 'editor': return { root: stateDir }
    default: return {}
  }
}

// 1) 逐插件提示词增量
host.services.prompt.section({ name: 'icl:simulation', order: 25, text: () => cfg.simulation })
const perPlugin = {}
let prev = host.services.prompt.assemble({})
perPlugin['(base: simulation)'] = chars(prev)
for (const id of DEFAULT_PLUGINS) {
  const { manifest, module } = await loadPlugin(id, { root })
  host.registerPlugin(manifest, module, { config: configFor(id) })
  try { await host.enable(id) } catch (error) { out[`enable:${id}`] = String(error?.message ?? error) }
  const now = host.services.prompt.assemble({})
  perPlugin[id] = chars(now) - chars(prev)
  prev = now
}
const assembled = host.services.prompt.assemble({})
out.systemPrompt = { totalChars: chars(assembled), totalBytes: bytes(assembled), perPluginAddChars: perPlugin }
out.sectionNames = host.services.prompt.names()
out.worldBrief = host.shared.get('world')?.renderBrief?.() ?? null

// 2) 工具面
const defs = host.services.tools.list()
const byPlugin = {}
for (const d of defs) {
  const row = byPlugin[d.plugin] ?? { count: 0, descriptionBytes: 0, paramBytes: 0 }
  row.count += 1
  row.descriptionBytes += bytes(d.description)
  row.paramBytes += bytes(JSON.stringify(d.parameters))
  byPlugin[d.plugin] = row
}
out.tools = {
  count: defs.length,
  descriptionBytes: defs.reduce((n, d) => n + bytes(d.description), 0),
  paramBytes: defs.reduce((n, d) => n + bytes(JSON.stringify(d.parameters)), 0),
  schemaBytes: defs.reduce((n, d) => n + bytes(JSON.stringify(d)), 0),
  byPlugin,
}

// 3) 前缀稳定性：首差异字符位置（= 前缀失效面）
const diffAt = (a, b) => { for (let i = 0; i < Math.min(a.length, b.length); i++) if (a[i] !== b[i]) return i; return a.length === b.length ? -1 : Math.min(a.length, b.length) }

const probes = {}

  const text = host.services.prompt.assemble({})

}
host.services.clock.set(() => T0)
const lease = host.services.focus.enter({ label: 'act:probe', title: '下矿', owner: 'focus', reason: '复算' })
const focusText = host.services.prompt.assemble({})
probes['with-focus'] = { totalChars: chars(focusText), totalBytes: bytes(focusText), firstDiffVsBase: diffAt(base, focusText) }
for (const [label, offset] of [['+5min-focus', 300_000], ['+1h-focus', 3_600_000]]) {
  host.services.clock.set(() => T0 + offset)
  const text = host.services.prompt.assemble({})
  probes[label] = { changed: text !== focusText, firstDiffVsBase: diffAt(base, text) }
}
lease.release()
host.services.clock.set(() => T0)
out.stability = { baseBytes: bytes(base), probes }

// 4) 每轮追加：eager-recall（prompt.onTurn）
const mem = host.shared.get('memory')
mem.put({ namespace: 'event', kind: 'note', key: 'n1', text: '下午在河边看到一只白鹭，站了很久。', weight: 1.5 })
mem.put({ namespace: 'event', kind: 'note', key: 'n2', text: '用户说想吃酸菜鱼，明天去买鱼。', weight: 1.5 })
const turn = await host.services.prompt.turnText({ message: '今天在河边走了走，你在干嘛' })
out.turnText = { chars: chars(turn), bytes: bytes(turn), text: turn }

// 5) 每轮追加：compaction 摘要（默认插件列表里已无 compaction → 期望 0）
out.contextAssemble = { chars: chars(await host.services.context.assemble({ history: [{ role: 'user', text: 'x' }], message: 'y' })) }
out.opts = { plugins: [...DEFAULT_PLUGINS].length, stateDir }

console.log(JSON.stringify(out, null, 1))
process.exit(0)
```

### 附录 B `icl-audit-gate.mjs`（睡眠/静音 gate 行为）

```js
// 复算脚本 B：睡眠/静音 gate 行为（虚拟时钟，不联网，不占端口）。
// 用法：node %TEMP%\icl-audit-gate.mjs C:\DSH\icl-wt\<worktree>
// 合并后验证：睡眠期 turn-end 不进 backlog；外部消息被拦；焦点摘要不直塞；醒来释放积压；静音锁。
import fs from 'node:fs'
import os from 'node:os'
import path from 'node:path'
import { pathToFileURL } from 'node:url'
const root = process.argv[2] || process.cwd()
const { createHost } = await import(pathToFileURL(path.join(root, 'packages/core/src/index.mjs')).href)
const timeMod = await import(pathToFileURL(path.join(root, 'packages/plugins/time/src/index.mjs')).href)
const manifest = JSON.parse(fs.readFileSync(path.join(root, 'packages/plugins/time/manifest.json'), 'utf8'))
const stateDir = fs.mkdtempSync(path.join(os.tmpdir(), 'icl-lock-check-'))
let clock = Date.parse('2026-11-05T23:00:00+08:00')
const logs = []
const host = createHost({ log: (e) => logs.push(e), now: () => clock })
host.registerPlugin(manifest, timeMod, { config: { stateFile: path.join(stateDir, 'time.json'), tickMs: 1000 } })
await host.enable('time')
const t = host.shared.get('time')
const out = {}
const pending = () => host.services.delivery.pendingCount()
const drainText = () => host.services.delivery.drain('session').map((b) => b.text)

// 1) 睡觉
await t.sleep({ message: '睡了', durationMin: 3 })
await new Promise((r) => setTimeout(r, 1500))
out.phaseAfterSleep = t.status().phase

// 2) 睡中 emit 事件（turn-end）：不该被拦、不该进积压
const before = t.status().backlog
const e1 = host.services.signals.emit({ id: 'e1', from: { plugin: 'dsh' }, kind: 'event', text: 'turn-end', tags: ['turn-end'] })
out.turnEnd = { accepted: e1.accepted, blocked: e1.blocked ?? false, backlogBefore: before, backlogAfter: t.status().backlog, pending: pending() }

// 3) 睡中外部消息（to:model）：应被拦 + 进 time 积压（有界）
const x1 = host.services.signals.offer({ id: 'x1', from: { plugin: 'qq' }, kind: 'content', text: '群里有人说话', focus: 'qq:group:1' })
out.externalDuringSleep = { accepted: x1.accepted, blocked: x1.blocked ?? false, reason: x1.reason, backlog: t.status().backlog, pending: pending() }

// 4) 睡中释放 focus 租约：焦点积压摘要不得直塞会话（改走 signals → 被 gate 拦）
const lease = host.services.focus.enter({ label: 'act:x', title: '写东西', owner: 'focus' })
host.services.focus.defer({ focus: 'qq:group:9', text: '别处的事', from: { plugin: 'qq' } })
lease.release()
await new Promise((r) => setTimeout(r, 100))
out.focusReleaseDuringSleep = { pending: pending(), backlog: t.status().backlog, drained: drainText().length }

// 5) 醒来：积压合成一条摘要释放（只一次，不逐条）
t.wake?.()
clock += 1000
await new Promise((r) => setTimeout(r, 1500))
const afterWake = drainText()
out.wake = { phase: t.status().phase, batches: afterWake.length, text: afterWake.map((x) => x.slice(0, 90)), backlog: t.status().backlog }

// 6) 静音锁：外部消息进积压，闹钟（admin/alarm）按配额穿透，解除后释放
const mute = t.mute({ minutes: 5, reason: '测试' })
out.mute = { muted: mute.muted, until: Boolean(mute.until) }
const m1 = host.services.signals.offer({ id: 'm1', from: { plugin: 'qq' }, kind: 'content', text: '静音期间的消息' })
out.mutedSignal = { accepted: m1.accepted, blocked: m1.blocked ?? false, reason: m1.reason, backlog: t.status().backlog }
const m2 = host.services.signals.offer({ id: 'm2', from: { plugin: 'mc' }, kind: 'event', text: '低血', tags: ['mc', 'safety', 'alarm'] })
out.mutedAlarm = { accepted: m2.accepted, blocked: m2.blocked ?? false, reason: m2.reason }
const un = t.unmute('测试结束')
out.unmute = { muted: un.muted, released: un.released }
const afterUnmute = drainText()
out.afterUnmute = { batches: afterUnmute.length, text: afterUnmute.map((x) => x.slice(0, 80)) }

// 7) 工具面：9 个 time 工具
out.tools = host.services.tools.names().filter((n) => n.startsWith('time_')).sort()
out.stats = host.services.signals.stats()
out.lock = t.status().lock ?? null
console.log(JSON.stringify(out, null, 1))
process.exit(0)
```
